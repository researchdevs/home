1. Debit Rs. 2000 if minimum balance after debit is Rs. 500.

CREATE TABLE accounts (
    account_id VARCHAR2(10) PRIMARY KEY,
    name       VARCHAR2(30),
    balance    NUMBER(10,2)
);

INSERT INTO accounts VALUES ('AC001', 'ANUJ', 5000);
INSERT INTO accounts VALUES ('AC002', 'MITA', 2400);
COMMIT;

SET SERVEROUTPUT ON;

DECLARE
    v_account_id accounts.account_id%TYPE := '&account_id';
    v_balance    accounts.balance%TYPE;
    v_debit      CONSTANT NUMBER := 2000;
BEGIN
    SELECT balance
    INTO v_balance
    FROM accounts
    WHERE account_id = v_account_id
    FOR UPDATE;

    IF v_balance - v_debit >= 500 THEN
        UPDATE accounts
        SET balance = balance - v_debit
        WHERE account_id = v_account_id;

        COMMIT;
        DBMS_OUTPUT.PUT_LINE('Debit successful.');
        DBMS_OUTPUT.PUT_LINE('New balance = ' || (v_balance - v_debit));
    ELSE
        DBMS_OUTPUT.PUT_LINE('Debit not allowed. Minimum balance of Rs. 500 must be maintained.');
    END IF;

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('Account not found.');
END;

2. Calculate area for radius 3 to 7 and store in AREA table.

CREATE TABLE area (
    radius NUMBER(5,2),
    area   NUMBER(10,2)
);

DECLARE
    v_area NUMBER;
BEGIN
    FOR r IN 3..7 LOOP
        v_area := ACOS(-1) * r * r;
        INSERT INTO area VALUES (r, ROUND(v_area, 2));
    END LOOP;
    COMMIT;
END;

3. Factorial of a number entered by user.

DECLARE
    v_n    NUMBER := &n;
    v_fact NUMBER := 1;
BEGIN
    IF v_n < 0 OR v_n != TRUNC(v_n) THEN
        DBMS_OUTPUT.PUT_LINE('Enter a non-negative integer.');
    ELSE
        FOR i IN 1..v_n LOOP
            v_fact := v_fact * i;
        END LOOP;
        DBMS_OUTPUT.PUT_LINE('Factorial = ' || v_fact);
    END IF;
END;

4. Insert an EMP record, update Blake by 2000 and Clark by 1500.
   If total salary exceeds 20000, undo only Blake/Clark updates.

-- Example table for this exercise
CREATE TABLE emp_plsql (
    emp_no   VARCHAR2(10) PRIMARY KEY,
    emp_name VARCHAR2(30),
    sal      NUMBER(10,2)
);

INSERT INTO emp_plsql VALUES ('E001','Harry',5000);
INSERT INTO emp_plsql VALUES ('E002','Blake',1000);
INSERT INTO emp_plsql VALUES ('E003','Jack',5000);
INSERT INTO emp_plsql VALUES ('E004','Clark',1000);
COMMIT;

DECLARE
    v_total NUMBER;
BEGIN
    INSERT INTO emp_plsql VALUES ('E005', 'David', 5000);

    SAVEPOINT before_salary_updates;

    UPDATE emp_plsql
    SET sal = sal + 2000
    WHERE emp_name = 'Blake';

    UPDATE emp_plsql
    SET sal = sal + 1500
    WHERE emp_name = 'Clark';

    SELECT SUM(sal) INTO v_total FROM emp_plsql;

    IF v_total > 20000 THEN
        ROLLBACK TO before_salary_updates;
        DBMS_OUTPUT.PUT_LINE('Total salary > 20000. Blake and Clark updates rolled back.');
    ELSE
        COMMIT;
        DBMS_OUTPUT.PUT_LINE('Updates committed.');
    END IF;
END;
/
